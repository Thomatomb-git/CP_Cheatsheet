% Algoritma/Fungsi: Definisi fungsi Mobius \mu(n) dan prekomputasi menggunakan Linear Sieve O(N).
% Kompleksitas Waktu: O(N) linier.
For any positive integer $n$,
\[
\mu(n) = 
\begin{cases} 
1 & \text{if } n = 1, \\
0 & \text{if } a^2 \mid n \text{ for some } a > 1, \\
(-1)^k & \text{if } n \text{ is the product of } k \text{ distinct primes. } \square
\end{cases}
\]
\begin{lstlisting}[language=c++]
  const int N = 1e6;
  int mobius[N + 1];
  vector<int> primes;
  bool is_prime[N + 1];

  // Linear Sieve O(N)
  void precompute_mobius() {
    fill(is_prime, is_prime + N + 1, true);
    is_prime[0] = is_prime[1] = false;
    mobius[1] = 1;

    for (int i = 2; i <= N; i++) {
      if (is_prime[i]) {
        primes.push_back(i);
        mobius[i] = -1;
      }
      for (int p : primes) {
        if (i * p > N) break;
        is_prime[i * p] = false;
        if (i % p == 0) {
          mobius[i * p] = 0;
          break;
        } else {
          mobius[i * p] = -mobius[i];
        }
      }
    }
  }
\end{lstlisting}
